# 1. Fresh clone simulation
cp .env.example .env

# 2. One-command setup
make setup
# This runs: docker compose up -d (infra) → pip install → build_workers.sh

# 3. Start dev
make dev          # backend on :8000
make frontend     # frontend on :3000 (in another terminal)

# 4. Or run everything in Docker
docker compose up -d
# Backend: http://localhost:8000
# Frontend: http://localhost:3000
# RabbitMQ UI: http://localhost:15672
```

**Available make commands:**
```
make help              # Show all commands
make setup             # Full setup (infra + venv + tools)
make dev               # Run backend (dev mode)
make frontend          # Run frontend (dev mode)
make build             # Build Docker images
make up                # Start everything in Docker
make down              # Stop everything
make db-reset          # Drop all tables
make db-shell          # Open PostgreSQL shell
make clean             # Remove venv, node_modules, cache
